select
    pb.TITLE_ID as "Book ID",
    pb.TITLE as "Book Title",
    pb.CONTRACT as "Contract Indicator",
    pb.PRICE as "Unit Price"
from 
    pb.TITLES pb
where 
    pb.CONTRACT > 0 AND pb.PRICE >= 30 AND pb.TOTAL_SALES > 0
order by
    pb.PRICE asc;
