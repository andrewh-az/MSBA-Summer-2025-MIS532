select
    ts.TITLE as "Book Title",
    ps.PUB_NAME as "Publisher Name",
    ps.CITY as "Publisher City"
from
    pb.TITLES ts
join
    pb.PUBLISHERS ps 
        ON ps.PUB_ID = ts.PUB_ID
where
    ts.TOTAL_SALES < 1000
    AND
    ps.STATE = 'AZ';
