select
    coalesce(pub.PUB_NAME, '-') as "Publisher Name",
    coalesce(ts.TITLE, '-') as "Book",
    count(DISTINCT sd.ORD_NUM) as "Number of Orders"
from
    pb.TITLES ts
left join
    pb.PUBLISHERS pub on pub.PUB_ID = ts.PUB_ID
left join
    pb.SALESDETAIL sd on sd.TITLE_ID = ts.TITLE_ID
group by
    rollup (pub.PUB_NAME, ts.TITLE)
order by
    pub.PUB_NAME, ts.TITLE
;
