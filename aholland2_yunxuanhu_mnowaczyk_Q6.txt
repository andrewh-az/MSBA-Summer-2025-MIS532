With Ranking AS (
select
    ts.TITLE_ID as "Title ID",
    ts.TITLE as "Book Title",
    coalesce (ts.PRICE, 0) as "Price",
    CASE
        when ts.PRICE is not NULL
        then RANK() over (order by ts.PRICE)
        else NULL
    END as "Rank",
    CASE
        when ts.PRICE is not NULL
        then DENSE_RANK () over (order by ts.PRICE)
        else NULL
    END as "DenseRank"
from
    pb.TITLES ts
inner join
    pb.SALESDETAIL sd on sd.TITLE_ID = ts.TITLE_ID
group by
    ts.TITLE_ID, 
    ts.TITLE, 
    ts.PRICE
)
select 
    *
from
    Ranking
where
    "DenseRank" <= 5
;
